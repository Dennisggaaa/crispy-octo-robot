local function make(class, properties, parent)
    local object = Instance.new(class)
    if object:IsA("GuiObject") then
        object.BorderSizePixel = 0
        object.BackgroundTransparency = 1
    end
    if object:IsA("TextLabel") or object:IsA("TextButton") then
        object.Text = ""
        object.FontFace = fonts.normal
        object.TextSize = 14
        object.TextColor3 = colors.text
        object.TextXAlignment = Enum.TextXAlignment.Left
        object.AutoLocalize = false
    end
    if object:IsA("GuiButton") then
        object.AutoButtonColor = false
    end
    for key, value in pairs(properties or {}) do
        object[key] = value
    end
    object.Parent = parent
    return object
end

local function corner(parent, radius)
    return make("UICorner", { CornerRadius = radius }, parent)
end

local function stroke(parent, transparency)
    make("UIStroke", {
        Color = Color3.new(1, 1, 1),
        Transparency = transparency,
        Thickness = 1,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
    }, parent)
end

local function padding(parent, left, right, top, bottom)
    make("UIPadding", {
        PaddingLeft = UDim.new(0, left),
        PaddingRight = UDim.new(0, right),
        PaddingTop = UDim.new(0, top),
        PaddingBottom = UDim.new(0, bottom),
    }, parent)
end

local function row(parent, gap, alignment)
    return make("UIListLayout", {
        FillDirection = Enum.FillDirection.Horizontal,
        SortOrder = Enum.SortOrder.LayoutOrder,
        Padding = UDim.new(0, gap),
        VerticalAlignment = Enum.VerticalAlignment.Center,
        HorizontalAlignment = alignment or Enum.HorizontalAlignment.Left,
    }, parent)
end

local function image(parent, key, size, order, transparency)
    return make("ImageLabel", {
        Size = UDim2.fromOffset(size, size),
        Image = "rbxassetid://" .. icons[key],
        ScaleType = Enum.ScaleType.Fit,
        LayoutOrder = order or 0,
        ImageTransparency = transparency or 0,
    }, parent)
end

local function animate(object, properties, duration)
    TweenService:Create(
        object,
        TweenInfo.new(duration or 0.14, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
        properties
    ):Play()
end

local function updateHighlight()
    if not selected or not gui.Parent then return end
    local scale = menuScale.Scale
    local delta = (selected.button.AbsolutePosition - tabRow.AbsolutePosition) / scale
    animate(highlight, {
        Position = UDim2.fromOffset(delta.X, delta.Y),
        Size = UDim2.fromOffset(
            selected.button.AbsoluteSize.X / scale,
            selected.button.AbsoluteSize.Y / scale
        ),
    }, 0.24)
end

local function drawTab(tab)
    local active = tab == selected
    tab.label.FontFace = active and fonts.bold or fonts.normal
    animate(tab.label, {
        TextColor3 = active and colors.text or (tab.hover and colors.sub or colors.dim),
    })
    animate(tab.icon, {
        ImageColor3 = active and colors.accent or (tab.hover and colors.sub or colors.dim),
    })
    tab.page.Visible = active
end

        BackgroundTransparency = enabled and 0.15 or 0.6,
    }, track)
    corner(knob, UDim.new(1, 0))

    local hit = make("TextButton", { Size = UDim2.fromScale(1, 1), ZIndex = 5 }, frame)
    local key = (frame.Parent and frame.Parent.Parent and frame.Parent.Parent.Name or "?")
        .. "/" .. tostring(frame.Position.Y.Offset)
    uiState[key] = on

    connect(hit.Activated, function()
        on = not on
        uiState[key] = on
        animate(track, {
            BackgroundColor3 = on and colors.accent or Color3.new(1, 1, 1),
            BackgroundTransparency = on and 0.35 or 0.86,
        })
        animate(knob, {
            Position = UDim2.new(0, on and 14 or 2, 0.5, 0),
            BackgroundColor3 = on and colors.accent or colors.sub,
            BackgroundTransparency = 0.15,
        })
        if type(onChanged) == "function" then pcall(onChanged, on) end
    end)
end

local function previewSlider(frame, text, onChanged)
    local area = make("Frame", {
        AnchorPoint = Vector2.new(1, 0),
        Position = UDim2.new(1, -10, 0, 0),
        Size = UDim2.new(0.47, 0, 1, 0),
    }, frame)

    local number = make("Frame", {
        Position = UDim2.new(0, 0, 0.5, -10),
        Size = UDim2.new(0.45, 0, 0, 20),
        BackgroundColor3 = Color3.new(1, 1, 1),
        BackgroundTransparency = 0.97,
    }, area)
    corner(number, UDim.new(0, 5))
    stroke(number, 0.96)

    local valueLabel = make("TextLabel", {
        Text = text,
        TextSize = 12,
        TextColor3 = colors.faint,
        TextXAlignment = Enum.TextXAlignment.Center,
        Size = UDim2.fromScale(1, 1),
    }, number)

    local track = make("Frame", {
        Position = UDim2.new(0.51, 0, 0.5, -1.5),
        Size = UDim2.new(0.49, 0, 0, 3),
        BackgroundColor3 = colors.faint,
        BackgroundTransparency = 0.3,
    }, area)
    corner(track, UDim.new(1, 0))

    local dot = make("Frame", {
        AnchorPoint = Vector2.new(0, 0.5),
        Position = UDim2.fromScale(0, 0.5),
        Size = UDim2.fromOffset(10, 10),
        BackgroundColor3 = colors.sub,
        BackgroundTransparency = 0.35,
    }, track)
    corner(dot, UDim.new(1, 0))

    local cfg = RANGES[text] or {
        min = 0, max = 100, suffix = "",
        default = tonumber(string.match(text, "%d+")) or 0,
    }
    local minimum = cfg.min
    local maximum = cfg.max
    local suffix = cfg.suffix
    local defaultValue = cfg.default
    local key = (frame.Parent and frame.Parent.Parent and frame.Parent.Parent.Name or "?")
        .. "/" .. tostring(frame.Position.Y.Offset)
    local lastValue = nil

    local function setValue(value)
        value = math.clamp(math.floor(value + 0.5), minimum, maximum)
        uiState[key] = value
        valueLabel.Text = tostring(value) .. suffix
        dot.Position = UDim2.fromScale((value - minimum) / math.max(1, maximum - minimum), 0.5)
        dot.AnchorPoint = Vector2.new(0.5, 0.5)
        if type(onChanged) == "function" and value ~= lastValue then
            lastValue = value
            pcall(onChanged, value)
        end
    end

    local function update(x)
        setValue((x - track.AbsolutePosition.X) / math.max(1, track.AbsoluteSize.X)
            * (maximum - minimum) + minimum)
    end

    local hit = make("TextButton", {
        Position = UDim2.new(0.51, -8, 0, 0),
        Size = UDim2.new(0.49, 16, 1, 0),
        ZIndex = 5,
    }, area)

    connect(hit.InputBegan, function(input)
        if input.UserInputType ~= Enum.UserInputType.MouseButton1
            and input.UserInputType ~= Enum.UserInputType.Touch then return end
        gesture = nil
        activeSlider = { input = input, update = update, scroll = previewScroll }
        previewScroll.ScrollingEnabled = false
        update(input.Position.X)
    end)

    setValue(defaultValue)
    return setValue
end

local function layoutScrollPanels(scroll, list, parentWidth)
    local narrow = parentWidth < 620
    local columnWidth = (parentWidth - 36) / 3
    local y = 10
    local maxY = 0

    for index, panel in ipairs(list) do
        panel.Position = narrow
            and UDim2.fromOffset(10, y)
            or UDim2.fromOffset(10 + (index - 1) * (columnWidth + 8), 10)

        panel.Size = UDim2.fromOffset(
            narrow and parentWidth - 20 or columnWidth,
            panel.Size.Y.Offset
        )

        local panelBottom = panel.Position.Y.Offset + panel.Size.Y.Offset
        if panelBottom > maxY then maxY = panelBottom end
        if narrow then y = y + panel.Size.Y.Offset + 12 end
    end

    scroll.CanvasSize = UDim2.fromOffset(0, math.max(maxY + 20, 300))
end

local function layoutPreview()
    local width = window.Size.X.Offset
    local narrow = width < 620
    local columnWidth = (width - 36) / 3
    local y = 10

    for index, panel in ipairs(panels) do
        panel.Position = narrow
            and UDim2.fromOffset(10, y)
            or UDim2.fromOffset(10 + (index - 1) * (columnWidth + 8), 10)

        panel.Size = UDim2.fromOffset(
            narrow and width - 20 or columnWidth,
            panel.Size.Y.Offset
        )

        y = y + panel.Size.Y.Offset + 12
    end

    previewScroll.CanvasSize = UDim2.fromOffset(0, math.max(y + 20, 300))

    pcall(layoutScrollPanels, visualsScroll, visualsPanels, width)
    pcall(layoutScrollPanels, triggerScroll, triggerPanels, width)
    pcall(layoutScrollPanels, miscScroll, miscPanels, width)
    pcall(layoutScrollPanels, settingsScroll, settingsPanels, width)
end

local function clampWindow()
    local size = window.AbsoluteSize
    local position = window.Position
    window.Position = UDim2.fromOffset(
        math.clamp(position.X.Offset, safeOrigin.X, safeOrigin.X + math.max(0, safeSize.X - size.X)),
        math.clamp(position.Y.Offset, safeOrigin.Y, safeOrigin.Y + math.max(0, safeSize.Y - size.Y))
    )
end

local function layoutNavigation()
    local width = window.Size.X.Offset
    local narrow = width < 620
    local cellHeight = 26
    local navHeight = narrow and (cellHeight * 2 + 8) or (cellHeight + 6)

    nav.Size = UDim2.fromOffset(narrow and (width - 20) or 0, navHeight)
    nav.AutomaticSize = narrow and Enum.AutomaticSize.None or Enum.AutomaticSize.X
    navCorner.CornerRadius = UDim.new(0, narrow and 12 or math.floor(navHeight / 2))
    body.Size = UDim2.new(1, 0, 1, -(41 + navHeight + 17))
    tabRow.Size = UDim2.fromOffset(narrow and (width - 26) or 0, navHeight - 6)
    tabRow.AutomaticSize = narrow and Enum.AutomaticSize.None or Enum.AutomaticSize.X

    local useGrid = navigationLayout:IsA("UIGridLayout")
    if narrow ~= useGrid then
        navigationLayout:Destroy()
        if narrow then
            navigationLayout = make("UIGridLayout", {
                SortOrder = Enum.SortOrder.LayoutOrder,
                FillDirection = Enum.FillDirection.Horizontal,
                FillDirectionMaxCells = 3,
            }, tabRow)
        else
            navigationLayout = row(tabRow, 2)
        end
    end

local detectTeamGroup

local TEAM_ATTRS = {
    "Team", "team", "TeamName", "TeamId", "Side", "Faction",
    "Alignment", "PlayerTeam",
}
local TEAM_VALUE_NAMES = { "Team", "TeamValue", "Faction", "Side", "TeamId" }

local function findTeamGroup(model)
    if not model then return nil end
    local node = model.Parent
    local depth = 0
    while node and node ~= Workspace and depth < 10 do
        if node.Name == TEAM_GROUP_B then return "B" end
        if node.Name == TEAM_GROUP_A then return "A" end
        node = node.Parent
        depth = depth + 1
    end
    return nil
end

local function findCharacterInWorkspace(plrName)
    local a = Workspace:FindFirstChild(TEAM_GROUP_A)
    if a then
        local inner = a:FindFirstChild("IIoooIIl")
        if inner then
            local m = inner:FindFirstChild(plrName)
            if m then return m, "A" end
        end
        local m = a:FindFirstChild(plrName)
        if m then return m, "A" end
    end
    local b = Workspace:FindFirstChild(TEAM_GROUP_B)
    if b then
        local m = b:FindFirstChild(plrName)
        if m then return m, "B" end
    end
    return nil, nil
end

local function getCharacter(plr)
    if not plr then return nil end
    local c = plr.Character
    if c and c.Parent then return c end
    local wc = findCharacterInWorkspace(plr.Name)
    if wc then return wc end
    return c
end

    return nil
end

local function probeRegistry(reg)
    if type(reg) ~= "table" or type(reg.isFriendly) ~= "function" then return false end
    if reg.players ~= nil and type(reg.players) ~= "table" then return false end
    local okA, a = pcall(reg.isFriendly, player)
    local okB, b = pcall(reg.isFriendly, player)
    if not okA or not okB or a ~= true or b ~= true then return false end

    local counts = 0
    local enemies = 0
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= player then
            local okC, c = pcall(reg.isFriendly, plr)
            if okC then
                counts = counts + 1
                if c == false then enemies = enemies + 1 end
            end
        end
    end
    if counts == 0 then return true end
    return enemies > 0
end

    if type(filtergc) ~= "function" then return nil, nil end
    for _, keys in ipairs(TEAM_REG_KEYS) do
        local ok, reg = pcall(filtergc, "table", { Keys = keys }, true)
        if ok and probeRegistry(reg) then return reg, keys end
    end
    return nil, nil
end

end

local playerCache = {}

local function buildPlayerEntry(plr)
    local char = getCharacter(plr)
    if not char then return nil end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum then return nil end

    local team = detectTeamGroup(plr)

    local friendly = false
    if plr == player then
        friendly = true
    elseif team and getMyTeam() then
        friendly = team == getMyTeam()
    elseif teamRegistry then
        local ok, f = pcall(teamRegistry.isFriendly, plr)
        if ok and f == true then friendly = true end
    else
        local m = findAnyTeamTag(player)
        local t = findAnyTeamTag(plr)
        if m ~= nil and t ~= nil then friendly = (m == t) end
    end

    local root = char:FindFirstChild("HumanoidRootPart")
        or char:FindFirstChild("head")
        or char:FindFirstChild("Head")
        or char:FindFirstChild("Torso")
    local head = char:FindFirstChild("head") or char:FindFirstChild("Head")
    local torso = char:FindFirstChild("Torso") or char:FindFirstChild("UpperTorso")
    local lLeg = char:FindFirstChild("Left Leg") or char:FindFirstChild("LeftUpperLeg")
    local rLeg = char:FindFirstChild("Right Leg") or char:FindFirstChild("RightUpperLeg")
    local lArm = char:FindFirstChild("Left Arm") or char:FindFirstChild("LeftUpperArm")
    local rArm = char:FindFirstChild("Right Arm") or char:FindFirstChild("RightUpperArm")
    local weapon = nil
    local tool = char:FindFirstChildOfClass("Tool")
    if tool then weapon = tool.Name end

    return {
        plr = plr,
        char = char,
        hum = hum,
        team = team,
        friendly = friendly,
        root = root,
        head = head,
        torso = torso,
        lLeg = lLeg,
        rLeg = rLeg,
        lArm = lArm,
        rArm = rArm,
        weapon = weapon,
    }
end

local function refreshPlayerCache()
    local newCache = {}
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= player then
            local ok, entry = pcall(buildPlayerEntry, plr)
            if ok and entry then newCache[plr] = entry end
        end
    end
    playerCache = newCache
end

local function isTeammate(plr)
    if plr == player then return true end
    local entry = playerCache[plr]
    if entry then return entry.friendly end
    return false
end

local function getCharacterCached(plr)
    local entry = playerCache[plr]
    if entry and entry.char and entry.char.Parent then return entry end
    local newEntry = buildPlayerEntry(plr)
    if newEntry then playerCache[plr] = newEntry end
    return newEntry
end

local DRONE_MARKERS = {
    "propellers", "Propellers", "motors", "Motors",
    "RootAttachment", "rotors", "Rotors",
}
local DRONE_OWNER_ATTRS = {
    "player", "Player", "owner", "Owner", "userid", "UserId",
    "userId", "PlayerId", "PlayerName",
}
local DRONE_THROTTLE_ATTRS = { "throttle", "Throttle" }

local function hasMarker(model, depth)
    if not model or depth > 3 then return false end
    if model:IsA("Model") or model:IsA("Folder") then
        for _, name in ipairs(DRONE_MARKERS) do
            if model:FindFirstChild(name) then return true end
        end
        for _, child in ipairs(model:GetChildren()) do
            if child:IsA("Model") or child:IsA("Folder") then
                if hasMarker(child, depth + 1) then return true end
            end
        end
    end
    return false
end

local function resolvePlayer(value)
    if value == nil then return nil end
    local vt = typeof(value)
    if vt == "Instance" then
        if value:IsA("Player") then return value end
        if value:IsA("Model") then
            local plr = Players:GetPlayerFromCharacter(value)
            if plr then return plr end
        end
        return nil
    end
    if type(value) == "number" then
        return Players:GetPlayerByUserId(value)
    end
    if type(value) == "string" then
        local n = tonumber(value)
        if n then
            local p = Players:GetPlayerByUserId(n)
            if p then return p end
        end
        local direct = Players:FindFirstChild(value)
        if direct and direct:IsA("Player") then return direct end
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr.Name == value or plr.DisplayName == value then return plr end
        end
    end
    return nil
end

local function findOwner(model)
    for _, attr in ipairs(DRONE_OWNER_ATTRS) do
        local ok, v = pcall(function() return model:GetAttribute(attr) end)
        if ok and v ~= nil then return v end
    end

    local node = model.Parent
    local depth = 0
    while node and depth < 3 do
        if node:IsA("Model") or node:IsA("Folder") then
            for _, attr in ipairs(DRONE_OWNER_ATTRS) do
                local ok, v = pcall(function() return node:GetAttribute(attr) end)
                if ok and v ~= nil then return v end
            end
        end
        node = node.Parent
        depth = depth + 1
    end

    for _, child in ipairs(model:GetChildren()) do
        if child:IsA("ObjectValue") or child:IsA("StringValue") or child:IsA("NumberValue") then
            local lower = string.lower(child.Name)
            if string.find(lower, "owner", 1, true)
                or string.find(lower, "player", 1, true)
                or string.find(lower, "user", 1, true) then
                local ok, v = pcall(function() return child.Value end)
                if ok and v ~= nil then return v end
            end
        end
    end
    return nil
end

local function findFirstPart(model)
    if not model then return nil end
    for _, child in ipairs(model:GetChildren()) do
        if child:IsA("BasePart") then return child end
    end
    for _, child in ipairs(model:GetDescendants()) do
        if child:IsA("BasePart") then return child end
    end
    return nil
end

local function analyzeDrone(model)
    if not model or not model:IsA("Model") then return nil end
    if model:FindFirstChildOfClass("Humanoid") then return nil end
    if Players:GetPlayerFromCharacter(model) then return nil end

    local score = 0
    local nameMatch = false
    local hasOwner = false
    local hasThrottle = false
    local hasMarker = false
    local hasRoot = false

    local lower = string.lower(model.Name)
    if string.find(lower, "drone", 1, true) or string.find(lower, "fpv", 1, true) then
        score = score + 2
        nameMatch = true
    end

    for _, attr in ipairs(DRONE_OWNER_ATTRS) do
        local ok, v = pcall(function() return model:GetAttribute(attr) end)
        if ok and v ~= nil then score = score + 2; hasOwner = true; break end
    end

    for _, attr in ipairs(DRONE_THROTTLE_ATTRS) do
        local ok, v = pcall(function() return model:GetAttribute(attr) end)
        if ok and v ~= nil then score = score + 2; hasThrottle = true; break end
    end

    if hasMarker(model, 0) then score = score + 2; hasMarker = true end

    if model.PrimaryPart and model.PrimaryPart:FindFirstChild("RootAttachment") then
        score = score + 1
        hasMarker = true
        hasRoot = true
    end

    if nameMatch then score = score + 1 end

    local candidate = (hasOwner and hasThrottle and (hasMarker or hasRoot))
        or (nameMatch and hasThrottle)
        or (nameMatch and score >= 4)
        or (score >= 6)

    if not candidate then return nil end

    local ownerVal = findOwner(model)
    if ownerVal == nil then return nil end
    local owner = resolvePlayer(ownerVal)
    if not owner then return nil end

    local part = model.PrimaryPart or model:FindFirstChild("head")
        or model:FindFirstChild("Head") or findFirstPart(model)
    if not part or not part:IsA("BasePart") then return nil end

    return {
        model = model,
        part = part,
        owner = owner,
        score = score,
        friendly = isTeammate(owner),
    }
end

local function scanDrones()
    local found = {}
    local playerChars = {}
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr.Character then playerChars[plr.Character] = true end
        local wc = findCharacterInWorkspace(plr.Name)
        if wc then playerChars[wc] = true end
    end

    local queue = {}
    local processed = 0
    local maxNodes = 20000

    for _, child in ipairs(Workspace:GetChildren()) do
        if child:IsA("Model") or child:IsA("Folder") or child:IsA("BasePart") then
            table.insert(queue, child)
        end
    end

    local head = 1
    while head <= #queue and processed < maxNodes do
        local node = queue[head]
        head = head + 1
        processed = processed + 1

        if node and node.Parent then
            local isModel = false
            pcall(function() isModel = node:IsA("Model") end)
            local descends = false

            if isModel then
                if not playerChars[node] then
                    local info = nil
                    pcall(function() info = analyzeDrone(node) end)
                    if info then
                        table.insert(found, info)
                    else
                        descends = true
                    end
                end
            elseif node:IsA("Folder") then
                descends = true
            end

            if descends then
                local ok, kids = pcall(function() return node:GetChildren() end)
                if ok and kids then
                    for _, child in ipairs(kids) do
                        if child:IsA("Model") or child:IsA("Folder") or child:IsA("BasePart") then
                            table.insert(queue, child)
                        end
                    end
                end
            end
        end
    end

    return found
end

function runDroneScan()
    if droneScanRunning then return end
    droneScanRunning = true
    task.spawn(function()
        local ok, result = pcall(scanDrones)
        if ok and type(result) == "table" then
            droneCache = result
            droneLastScan = os.clock()
        end
        droneScanRunning = false
    end)
end

local function getRaycastParams()
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    local filter = {}
    if player.Character then table.insert(filter, player.Character) end
    if camera then table.insert(filter, camera) end
    params.FilterDescendantsInstances = filter

local function isPartVisible(part)
    if not part or not camera then return true end
    local origin = camera.CFrame.Position
    local delta = part.Position - origin
    if delta.Magnitude < 0.5 then return true end

    local params = getRaycastParams()
    local wasBusy = aimBusy
    aimBusy = true
    local ok, result = pcall(function()
        return Workspace:Raycast(origin, delta, params)
    end)
    aimBusy = wasBusy

    if not ok or not result then return true end
    local hit = result.Instance
    if hit == part then return true end
    local parent = part.Parent
    if parent and hit:IsDescendantOf(parent) then return true end
    return false
end

    return entry.head or entry.root or entry.torso
end

        if plr ~= player then
            local entry = getCharacterCached(plr)
            if entry and entry.hum and entry.hum.Health > 0 then
                if not (Config.Aim.TeamCheck and isTeammate(plr)) then
                    local part = getHitboxPart(entry, hitboxPref)
                    if part and part:IsA("BasePart") then
                        local worldPos = part.Position
                        if Config.Aim.Prediction and Config.Aim.Smoothing > 0 then
                            local vel = entry.root and entry.root.AssemblyLinearVelocity or Vector3.zero
                            worldPos = worldPos + vel * 0.1
                        end
                        local dist = (worldPos - camPos).Magnitude
                        if dist <= maxDist then
                            local okp, screen = pcall(function()
                                return camera:WorldToViewportPoint(worldPos)
                            end)
                            if okp and screen and screen.Z > 0.05 then
                                local dx = screen.X - cx
                                local dy = screen.Y - cy
                                local sd = math.sqrt(dx * dx + dy * dy)
                                if sd < fov then
                                    local visible = true
                                    if Config.Aim.VisibleCheck or Config.Aim.WallCheck then
                                        visible = isPartVisible(part)
                                    end
                                    if visible then
                                        local priority = Config.Aim.Priority
                                        local score = sd
                                        if priority == "distance" then score = dist end
                                        if priority == "hp" then
                                            score = math.clamp(1 - entry.hum.Health / math.max(1, entry.hum.MaxHealth), 0, 1) * 1000
                                                + sd
                                        end
                                        if score < bestScore then
                                            bestScore = score
                                            bestPoint = worldPos
                                            bestName = plr.Name
                                        end
                                    end
                                end
                            end
                        end
                    end
                end
            end
        end
    end

    if Config.Aim.FpvDrones then
        for _, drone in ipairs(droneCache) do
            local model = drone.model
            if model and model.Parent and drone.part and drone.part.Parent then
                local skip = false
                if Config.Aim.TeamCheck and drone.owner and isTeammate(drone.owner) then
                    skip = true
                end
                if not skip then
                    local worldPos = drone.part.Position
                    local dist = (worldPos - camPos).Magnitude
                    if dist <= maxDist then
                        local flatDelta = Vector3.new(worldPos.X - camPos.X, 0, worldPos.Z - camPos.Z)
                        local behind = false
                        if flatDelta.Magnitude > 0.001 then
                            behind = flatLook:Dot(flatDelta.Unit) < -0.1
                        end

                        if behind then
                            if dist < Config.Aim.DroneRadius then
                                local score = -100000 - dist
                                if score < bestScore then
                                    bestScore = score
                                    bestPoint = worldPos
                                    bestName = model.Name
                                end
                            end
                        else
                            local okp, screen = pcall(function()
                                return camera:WorldToViewportPoint(worldPos)
                            end)
                            if okp and screen and screen.Z > 0.05 then
                                local dx = screen.X - cx
                                local dy = screen.Y - cy
                                local sd = math.sqrt(dx * dx + dy * dy)
                                if sd < fov then
                                    local visible = true
                                    if Config.Aim.VisibleCheck then
                                        visible = isPartVisible(drone.part)
                                    end
                                    if visible then
                                        local score = sd + 30
                                        if score < bestScore then
                                            bestScore = score
                                            bestPoint = worldPos
                                            bestName = model.Name
                                        end
                                    end
                                end
                            end
                        end
                    end
                end
            end
        end
    end

    if bestName then
        currentAimTargetName = bestName
    end
    currentAimTarget = bestPoint
    return bestPoint
end

local function installRaycastHook()
    if type(hookfunction) ~= "function" then return false end
    local ok = pcall(function()
        originalRaycast = hookfunction(Workspace.Raycast, function(self, origin, direction, params)
            if not Config.Aim.Enabled then
                return originalRaycast(self, origin, direction, params)
            end
            if type(checkcaller) == "function" then
                local ok2, isCaller = pcall(checkcaller)
                if ok2 and isCaller then
                    return originalRaycast(self, origin, direction, params)
                end
            end
            if aimBusy then
                return originalRaycast(self, origin, direction, params)
            end
            aimBusy = true
            local ok3, target = pcall(acquireTarget, origin)
            aimBusy = false
            if ok3 and target then
                local delta = target - origin
                if delta.Magnitude > 0.001 then
                    return originalRaycast(self, origin, delta.Unit * direction.Magnitude, params)
                end
            end
            return originalRaycast(self, origin, direction, params)
        end)
    end)
    return ok
end

local function installLookSpoof()
    if type(getrawmetatable) ~= "function" then return false end
    local ok = pcall(function()
        local mt = getrawmetatable(game)
        local orig = mt.__namecall

        setreadonly(mt, false)
        mt.__namecall = newcclosure(function(self, ...)
            local method = getnamecallmethod()
            if method == "FireServer" then
                if self == acRemote and Config.Antiban.SilenceRemote then
                    return
                end
                if self == terminateRemote and Config.Antiban.BlockTerminate then
                    return
                end
                if self == reportRemote and Config.Antiban.AntiReport then
                    return
                end

                if self == lookRemote and Config.Antiban.SpoofLook and Config.Aim.Enabled then
                    local args = { ... }
                    if currentAimTarget and camera then
                        local dir = (currentAimTarget - camera.CFrame.Position)
                        if dir.Magnitude > 0.001 then
                            for i = 1, #args do
                                if typeof(args[i]) == "Vector3" then
                                    args[i] = dir.Unit
                                end
                            end
                            return orig(self, table.unpack(args))
                        end
                    end
                end

                if Config.Antiban.HeartbeatJitter then
                    for _, bullet in ipairs(bulletRemotes) do
                        if self == bullet then
                            if math.random() < 0.02 then
                                task.wait(math.random() * 0.01)
                            end
                            break
                        end
                    end
                end
            end
            return orig(self, ...)
        end)
        setreadonly(mt, true)
        namaecallHooked = true
    end)
    return ok
end

local function applyAntiban()
    if Config.Antiban.SilenceRemote or Config.Antiban.BlockTerminate or Config.Antiban.SpoofLook
        or Config.Antiban.AntiReport or Config.Antiban.HeartbeatJitter then
        if not namaecallHooked then installLookSpoof() end
    end

    if Config.Antiban.MonitorRegistry then
        for _, entry in ipairs(registryRemotes) do
            pcall(function()
                local r = entry.instance
                if r:IsA("RemoteFunction") then
                    if not r.__monitored then
                        r.__monitored = true
                        r.OnClientInvoke = function(...)
                            if Config.Antiban.MonitorRegistry then
                                print("[AC] " .. entry.name .. " OnClientInvoke args:", ...)
                            end
                            return nil
                        end
                    end
                elseif r:IsA("RemoteEvent") then
                    if not r.__monitored then
                        r.__monitored = true
                        connect(r.OnClientEvent, function(...)
                            if Config.Antiban.MonitorRegistry then
                                print("[AC] " .. entry.name .. " OnClientEvent:", ...)
                            end
                        end)
                    end
                end
            end)
        end
    end
end

local triggerLastFire = 0
local function triggerTick()
    if not Config.Trigger.Enabled then return end

    local now = os.clock()
    local minDelay = Config.Trigger.Delay / 1000
    if Config.Antiban.RateLimitTrigger then
        minDelay = math.max(minDelay, 0.08)
    end
    if Config.Antiban.HeartbeatJitter then
        minDelay = minDelay + math.random() * 0.02
    end
    if now - triggerLastFire < minDelay then return end

    local char = player.Character
    if not char then return end
    local tool = char:FindFirstChildOfClass("Tool")
    if not tool then return end

    if not camera then return end
    local vp = camera.ViewportSize
    local cx = vp.X * 0.5
    local cy = vp.Y * 0.5
    local camPos = camera.CFrame.Position

    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= player then
            if not (Config.Trigger.TeamCheck and isTeammate(plr)) then
                local entry = getCharacterCached(plr)
                if entry and entry.hum and entry.hum.Health > 0 then
                    local part
                    if Config.Trigger.HeadshotOnly then
                        part = entry.head
                    else
                        part = getHitboxPart(entry, Config.Aim.Hitbox)
                    end
                    if part and part:IsA("BasePart") then
                        local dist = (part.Position - camPos).Magnitude
                        if dist <= Config.Trigger.Range then
                            local okp, screen = pcall(function()
                                return camera:WorldToViewportPoint(part.Position)
                            end)
                            if okp and screen and screen.Z > 0.05 then
                                local dx = screen.X - cx
                                local dy = screen.Y - cy
                                local sd = math.sqrt(dx * dx + dy * dy)
                                if sd < Config.Trigger.PixelThreshold then
                                    local visible = true
                                    if Config.Trigger.VisibleCheck then
                                        visible = isPartVisible(part)
                                    end
                                    if visible then
                                        triggerLastFire = now
                                        pcall(function() tool:Activate() end)
                                        return
                                    end
                                end
                            end
                        end
                    end
                end
            end
        end
    end
end

local function getBbox(cameraRef, model)
    local ok, cf, size = pcall(function()
        local c, s = model:GetBoundingBox()
        return c, s
    end)
    if not ok or not cf or not size then return nil end

    local sx = size.X * 0.5
    local sy = size.Y * 0.5
    local sz = size.Z * 0.5
    local corners = {
        cf * Vector3.new(-sx, -sy, -sz),
        cf * Vector3.new( sx, -sy, -sz),
        cf * Vector3.new( sx, -sy,  sz),
        cf * Vector3.new(-sx, -sy,  sz),
        cf * Vector3.new(-sx,  sy, -sz),
        cf * Vector3.new( sx,  sy, -sz),
        cf * Vector3.new( sx,  sy,  sz),
        cf * Vector3.new(-sx,  sy,  sz),
    }

    local minX, minY, maxX, maxY = math.huge, math.huge, -math.huge, -math.huge
    local any = false
    for i = 1, 8 do
        local okp, sp = pcall(function()
            return cameraRef:WorldToViewportPoint(corners[i])
        end)
        if okp and sp and sp.Z > 0.05 then
            any = true
            if sp.X < minX then minX = sp.X end
            if sp.X > maxX then maxX = sp.X end
            if sp.Y < minY then minY = sp.Y end
            if sp.Y > maxY then maxY = sp.Y end
        end
    end
    if not any then return nil end
    return minX, minY, maxX, maxY, corners
end

local function clampBox(minX, minY, maxX, maxY)
    local w = maxX - minX
    local h = maxY - minY
    if h < 8 then h = 8 end
    if w < h * 0.45 then w = h * 0.45 end
    if w > h * 2.2 then w = h * 2.2 end
    local cx = (minX + maxX) * 0.5
    local cy = (minY + maxY) * 0.5
    return cx - w * 0.5, cy - h * 0.5, w, h
end

local function smoothValue(last, current)
    if not last then return current end
    local ratio = current / math.max(0.001, last)
    if ratio > 2.5 or ratio < (1 / 2.5) then
        return last + (current - last) * 0.5
    end
    return current
end

local function renderPlayerSlot(slot, entry)
    if not camera then hidePlayerSlot(slot); return end
    if not entry.hum or entry.hum.Health <= 0 then
        if Config.Visuals.PlayerEsp.IgnoreDowned then
            slot.emptyFrames = slot.emptyFrames + 1
            if slot.emptyFrames >= 3 then hidePlayerSlot(slot) end
            return
        end
    end

    if Config.Visuals.PlayerEsp.TeamCheck and entry.friendly then
        hidePlayerSlot(slot)
        slot.emptyFrames = 0
        return
    end

    local root = entry.root
    if not root or not root:IsA("BasePart") then
        slot.emptyFrames = slot.emptyFrames + 1
        if slot.emptyFrames >= 3 then hidePlayerSlot(slot) end
        return
    end

    local dist = (root.Position - camera.CFrame.Position).Magnitude
    if dist > Config.Visuals.PlayerEsp.MaxDistance then
        slot.emptyFrames = slot.emptyFrames + 1
        if slot.emptyFrames >= 3 then hidePlayerSlot(slot) end
        return
    end

    local minX, minY, maxX, maxY, corners = getBbox(camera, entry.char)
    if not minX then
        slot.emptyFrames = slot.emptyFrames + 1
        if slot.emptyFrames >= 3 then hidePlayerSlot(slot) end
        return
    end

    slot.emptyFrames = 0

    local bx, by, bw, bh = clampBox(minX, minY, maxX, maxY)
    bw = smoothValue(slot.lastW, bw)
    bh = smoothValue(slot.lastH, bh)
    local cx = (minX + maxX) * 0.5
    local cy = (minY + maxY) * 0.5
    bx = cx - bw * 0.5
    by = cy - bh * 0.5

    slot.lastX = bx
    slot.lastY = by
    slot.lastW = bw
    slot.lastH = bh

    if Config.Visuals.PlayerEsp.FilledBox and slot.boxFill then
        safeSet(slot.boxFill, {
            Visible = true,
            Position = Vector2.new(bx, by),
            Size = Vector2.new(bw, bh),
            Color = Color3.new(1, 1, 1),
            Filled = true,
            Thickness = 0,
            Transparency = 0.85,

        safeSet(slot.boxFill, { Visible = false })
    end

            Visible = true,
            Position = Vector2.new(bx, by),
            Size = Vector2.new(bw, bh),

            Filled = false,

            Size = Vector2.new(3, barH),
            Color = Color3.new(0, 0, 0),
            Filled = true,
            Transparency = 0.3,

        safeSet(slot.distance, { Visible = false })

        end
    else
        for i = 1, 12 do
            if slot.skeleton[i] then safeSet(slot.skeleton[i], { Visible = false }) end
        end
    end

    slot.lastVisible = true
end

        distance = newDrawing("Text"),
        owner = newDrawing("Text"),
        tracer = newDrawing("Line"),
        lastVisible = false,
        emptyFrames = 0,
    }
    for i = 1, 12 do
        slot.lines[i] = newDrawing("Line")
    end
    droneEspSlots[model] = slot
    return slot
end

local function hideDroneSlot(slot)
    if not slot.lastVisible then return end
    for i = 1, 12 do safeSet(slot.lines[i], { Visible = false }) end
    safeSet(slot.name, { Visible = false })
    safeSet(slot.distance, { Visible = false })
    safeSet(slot.owner, { Visible = false })
    safeSet(slot.tracer, { Visible = false })
    slot.lastVisible = false
end

local EDGES = {
    { 1, 2 }, { 2, 3 }, { 3, 4 }, { 4, 1 },
    { 5, 6 }, { 6, 7 }, { 7, 8 }, { 8, 5 },
    { 1, 5 }, { 2, 6 }, { 3, 7 }, { 4, 8 },
}

local function renderDroneSlot(slot, drone)
    local model = drone.model
    if not model or not model.Parent then
        slot.emptyFrames = slot.emptyFrames + 1
        if slot.emptyFrames >= 3 then hideDroneSlot(slot) end
        return
    end
    if Config.Visuals.DroneEsp.TeamCheck and drone.owner and isTeammate(drone.owner) then
        hideDroneSlot(slot)
        slot.emptyFrames = 0
        return
    end
    if not camera then hideDroneSlot(slot); return end

    local part = drone.part
    if not part or not part.Parent then
        part = findFirstPart(model)
        if not part then
            slot.emptyFrames = slot.emptyFrames + 1
            if slot.emptyFrames >= 3 then hideDroneSlot(slot) end
            return
        end
    end

    local dist = (part.Position - camera.CFrame.Position).Magnitude
    if dist > Config.Visuals.DroneEsp.MaxDistance then
        slot.emptyFrames = slot.emptyFrames + 1
        if slot.emptyFrames >= 3 then hideDroneSlot(slot) end
        return
    end

    local ok, cf, size = pcall(function()
        local c, s = model:GetBoundingBox()
        return c, s
    end)
    if not ok or not cf then
        slot.emptyFrames = slot.emptyFrames + 1
        if slot.emptyFrames >= 3 then hideDroneSlot(slot) end
        return
    end

    slot.emptyFrames = 0

    local sx = size.X * 0.5
    local sy = size.Y * 0.5
    local sz = size.Z * 0.5
    local corners = {
        cf * Vector3.new(-sx, -sy, -sz),
        cf * Vector3.new( sx, -sy, -sz),
        cf * Vector3.new( sx, -sy,  sz),
        cf * Vector3.new(-sx, -sy,  sz),
        cf * Vector3.new(-sx,  sy, -sz),
        cf * Vector3.new( sx,  sy, -sz),
        cf * Vector3.new( sx,  sy,  sz),
        cf * Vector3.new(-sx,  sy,  sz),
    }

    local projected = {}
    for i = 1, 8 do
        local okp, sp = pcall(function()
            return camera:WorldToViewportPoint(corners[i])
        end)
        if okp and sp and sp.Z > 0.05 then
            projected[i] = Vector2.new(sp.X, sp.Y)
        end
    end

    local anyLine = false
    for i = 1, 12 do
        local edge = EDGES[i]
        local a = projected[edge[1]]
        local b = projected[edge[2]]
        local line = slot.lines[i]
        if line then
            if a and b and Config.Visuals.DroneEsp.Box then
                safeSet(line, {
                    Visible = true,
                    From = a,
                    To = b,
                    Color = Color3.new(1, 1, 1),
                    Thickness = 1,
                    Transparency = 1,
                })
                anyLine = true
            else
                safeSet(line, { Visible = false })
            end
        end
    end

    if not anyLine and Config.Visuals.DroneEsp.Box then
        slot.emptyFrames = slot.emptyFrames + 1
        if slot.emptyFrames >= 3 then hideDroneSlot(slot) end
        return
    end

    local cX, cY, cN = 0, 0, 0
    local minX, minY, maxX, maxY = math.huge, math.huge, -math.huge, -math.huge
    for i = 1, 8 do
        local p = projected[i]
        if p then
            cX = cX + p.X
            cY = cY + p.Y
            cN = cN + 1
            if p.X < minX then minX = p.X end
            if p.X > maxX then maxX = p.X end
            if p.Y < minY then minY = p.Y end
            if p.Y > maxY then maxY = p.Y end
        end
    end
    if cN > 0 then
        cX = cX / cN
        cY = cY / cN
    end

    if Config.Visuals.DroneEsp.Name and slot.name then
        safeSet(slot.name, {
            Visible = true,
            Text = model.Name,
            Color = Color3.new(1, 1, 1),
            Size = 13,
            Center = true,
            Outline = true,
            OutlineColor = Color3.new(0, 0, 0),
            Position = Vector2.new(cX, minY - 16),
            Transparency = 1,
        })
    else
        safeSet(slot.name, { Visible = false })
    end

    if Config.Visuals.DroneEsp.Distance and slot.distance then
        safeSet(slot.distance, {
            Visible = true,
            Text = string.format("%dm", math.floor(dist + 0.5)),
            Color = Color3.new(1, 1, 1),
            Size = 13,
            Center = true,
            Outline = true,
            OutlineColor = Color3.new(0, 0, 0),
            Position = Vector2.new(cX, maxY + 2),
            Transparency = 1,
        })
    else
        safeSet(slot.distance, { Visible = false })
    end

    if Config.Visuals.DroneEsp.Owner and slot.owner and drone.owner then
        safeSet(slot.owner, {
            Visible = true,
            Text = drone.owner.Name,
            Color = Color3.new(1, 1, 1),
            Size = 11,
            Center = true,
            Outline = true,
            OutlineColor = Color3.new(0, 0, 0),
            Position = Vector2.new(cX, maxY + 16),
            Transparency = 1,
        })
    else
        safeSet(slot.owner, { Visible = false })
    end

    if Config.Visuals.DroneEsp.Tracer and slot.tracer then
        local vp = camera.ViewportSize
        safeSet(slot.tracer, {
            Visible = true,
            From = Vector2.new(vp.X * 0.5, vp.Y),
            To = Vector2.new(cX, cY),
            Color = Color3.new(1, 1, 1),
            Thickness = 1,
            Transparency = 0.6,
        })
    else
        safeSet(slot.tracer, { Visible = false })
    end

    slot.lastVisible = true
end

        return

local flyBodyVel = nil
local flyBodyGyro = nil
local flyAnimConn = nil

local function startFly()
    local char = player.Character
    if not char then return end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hrp then return end

    pcall(function()
        flyBodyVel = Instance.new("BodyVelocity")
        flyBodyVel.MaxForce = Vector3.new(1e5, 1e5, 1e5)
        flyBodyVel.Velocity = Vector3.zero
        flyBodyVel.Parent = hrp

        flyBodyGyro = Instance.new("BodyGyro")
        flyBodyGyro.MaxTorque = Vector3.new(1e5, 1e5, 1e5)
        flyBodyGyro.P = 1e4
        flyBodyGyro.Parent = hrp
    end)

    if flyAnimConn then flyAnimConn:Disconnect() end
    flyAnimConn = RunService.Heartbeat:Connect(function()
        if not Config.Misc.Fly or not flyBodyVel or not flyBodyVel.Parent then return end
        local hrp2 = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
        if not hrp2 or not camera then return end

        local dir = Vector3.zero
        local move = UserInputService:GetKeysPressed()
        for _, k in ipairs(move) do
            if k.KeyCode == Enum.KeyCode.W then dir = dir + camera.CFrame.LookVector end
            if k.KeyCode == Enum.KeyCode.S then dir = dir - camera.CFrame.LookVector end
            if k.KeyCode == Enum.KeyCode.A then dir = dir - camera.CFrame.RightVector end
            if k.KeyCode == Enum.KeyCode.D then dir = dir + camera.CFrame.RightVector end
            if k.KeyCode == Enum.KeyCode.Space then dir = dir + Vector3.new(0, 1, 0) end
            if k.KeyCode == Enum.KeyCode.LeftControl then dir = dir - Vector3.new(0, 1, 0) end
        end
        if dir.Magnitude > 0.01 then dir = dir.Unit end

        pcall(function()
            flyBodyVel.Velocity = dir * Config.Misc.FlySpeed
            flyBodyGyro.CFrame = camera.CFrame
        end)
    end)
    table.insert(connections, flyAnimConn)
end

local function stopFly()
    if flyBodyVel then pcall(function() flyBodyVel:Destroy() end); flyBodyVel = nil end
    if flyBodyGyro then pcall(function() flyBodyGyro:Destroy() end); flyBodyGyro = nil end
    if flyAnimConn then pcall(function() flyAnimConn:Disconnect() end); flyAnimConn = nil end
end

local function applyNoclip(on)
    local function setCollide(node, c)
        for _, d in ipairs(node:GetDescendants()) do
            if d:IsA("BasePart") and d.CanCollide ~= c then
                d.CanCollide = c
            end
        end
    end
    local char = player.Character
    if not char then return end
    pcall(function() setCollide(char, not on) end)
end

local function antiFallTick()
    local char = player.Character
    if not char then return end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hrp then return end
    if hrp.Position.Y < -50 then
        pcall(function()
            hrp.CFrame = CFrame.new(0, 100, 0)
        end)
    end
end

local function charTick()
    local char = player.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum then return end

    if Config.Misc.SpeedEnabled then
        if hum.WalkSpeed ~= Config.Misc.WalkSpeed then
            hum.WalkSpeed = Config.Misc.WalkSpeed
        end
    end

    if Config.Misc.JumpEnabled then
        hum.JumpPower = Config.Misc.JumpPower
    end

    if Config.Auto.AlwaysSprint then
        if hum.WalkSpeed < 20 and not Config.Misc.SpeedEnabled then
            hum.WalkSpeed = 20
        end
    end

    if Config.Auto.AutoReload then
        local tool = char:FindFirstChildOfClass("Tool")
        if tool then
            if Config.Weapon.InstaReload then
                pcall(function()
                    if tool:FindFirstChild("Ammo") then
                        tool.Ammo.Value = 30
                    end
                end)
            end
        end
    end

    if Config.Misc.Freeze then
        hum.WalkSpeed = 0
        hum.JumpPower = 0
    elseif Config.Misc.SpeedEnabled then
        hum.WalkSpeed = Config.Misc.WalkSpeed
    elseif Config.Misc.JumpEnabled then
        hum.JumpPower = Config.Misc.JumpPower
    end
end

local autoJumpConn = nil
local function setupAutoJump()
    if autoJumpConn then
        pcall(function() autoJumpConn:Disconnect() end)
        autoJumpConn = nil
    end
    if Config.Misc.InfiniteJump or Config.Auto.AutoJump or Config.Auto.BunnyHop then
        autoJumpConn = UserInputService.JumpRequest:Connect(function()
            local char = player.Character
            if not char then return end
            local hum = char:FindFirstChildOfClass("Humanoid")
            if not hum then return end
            if Config.Misc.InfiniteJump then
                pcall(function()
                    hum:ChangeState(Enum.HumanoidStateType.Jumping)
                end)
            end
            if Config.Auto.BunnyHop or Config.Auto.AutoJump then
                pcall(function()
                    hum.Jump = true
                end)
                task.wait(0.05)
                pcall(function()
                    hum:ChangeState(Enum.HumanoidStateType.Jumping)
                end)
            end
        end)
        table.insert(connections, autoJumpConn)
    end
end
